% ============================================================
% FIG 9 — Decision Transition (flow-width diagram)
% Shows the 1,713 fusion-attributable decision changes
% from Config 4 (B3-only) to Config 5 (full stack).
% Arrow thickness proportional to message count.
% Data from Section VI-B and Fig 4 of the paper:
%   Total transitions: 1,713
%   Caution->Reject:  1,585  (92.5%)
%   Accept->Caution:    128   (7.5%)
%   Direct Accept<->Reject: 0
% Full-stack distribution (n=10,000):
%   Accept:  6,000  (60%)
%   Caution: 1,900  (19%)
%   Reject:  2,100  (21%)
% ============================================================
% PREAMBLE:
%   \usepackage{tikz}
%   \usepackage{xcolor}
%   \usetikzlibrary{arrows.meta, positioning, calc}
%
% PLACEMENT: single column
%   \begin{figure}[t]
%     \centering
%     \input{fig9_decision_transition.tex}
%     \caption{Fusion-attributable decision transitions (Config~4
%     $\rightarrow$ Config~5, $n=10{,}000$). Arrow width is proportional
%     to message count. Evidence fusion causes 1,713 real decision
%     changes: 1,585 Caution$\rightarrow$Reject escalations on genuine
%     attacks (92.5\%) and 128 Accept$\rightarrow$Caution additions
%     (7.5\%). Zero direct Accept$\leftrightarrow$Reject reversals occur.}
%     \label{fig:decision_transition}
%   \end{figure}
% ============================================================

\begin{tikzpicture}[
  every node/.style={font=\scriptsize},
]

\definecolor{accgreen}{RGB}{40,160,80}
\definecolor{cauorange}{RGB}{220,140,20}
\definecolor{rejred}{RGB}{190,40,40}
\definecolor{arrowgray}{RGB}{140,140,150}

% ============================================================
% LAYOUT
% Left column:  B3-only decisions
% Right column: Full-stack decisions
% ============================================================

% --- LEFT NODES (B3-only) ---
% Accept:  6,128 (B3 alone passes these; fusion adds 128 to Caution)
% Caution: 4,657 (B3 alone flags these; fusion escalates 1,585 to Reject)
% Reject:    515 (B3 alone rejects; fusion keeps)
% Note: numbers estimated from paper data for the B3-alone config

\def\lx{1.6}   % left column x
\def\rx{6.5}   % right column x
\def\boxW{1.8} % box width
\def\boxH{0.9} % base box height (scaled by count)

% Scale: \mscale unchanged to preserve proportional Sankey encoding
\def\mscale{0.00035}  % cm per message

% Left boxes — proportional heights from message counts
\pgfmathsetmacro{\accHl}{6128*\mscale}
\pgfmathsetmacro{\cauHl}{4657*\mscale}
% [LAYOUT] Clamp only the Reject box: proportional height (515*0.00035=0.18cm)
%          is too small to render the label legibly. We apply a minimum of
%          0.50cm so the text fits while \mscale and all other boxes remain
%          unchanged. This preserves the Sankey proportionality for the two
%          large boxes while fixing the tiny-box readability defect.
\pgfmathsetmacro{\rejHlRaw}{515*\mscale}
\pgfmathsetmacro{\rejHl}{max(\rejHlRaw, 0.50)}

% Right boxes (full stack)
\pgfmathsetmacro{\accHr}{6000*\mscale}
\pgfmathsetmacro{\cauHr}{1900*\mscale + 0.1}
\pgfmathsetmacro{\rejHr}{2100*\mscale}

% Y positions (bottom of each box), left column
% stack from bottom: Reject, Caution, Accept
\def\rejYl{0}
\pgfmathsetmacro{\cauYl}{\rejYl + \rejHl + 0.15}
\pgfmathsetmacro{\accYl}{\cauYl + \cauHl + 0.15}

% Y positions, right column — same order
\def\rejYr{0}
\pgfmathsetmacro{\cauYr}{\rejYr + \rejHr + 0.15}
\pgfmathsetmacro{\accYr}{\cauYr + \cauHr + 0.15}

% ============================================================
% DRAW LEFT BOXES
% ============================================================
% Accept (left)
\fill[accgreen!20, draw=accgreen, thick, rounded corners=2pt]
  (\lx, \accYl) rectangle +(\boxW, \accHl);
\node[font=\scriptsize\bfseries, text=accgreen, align=center]
  at (\lx + 0.5*\boxW, \accYl + 0.5*\accHl)
  {Accept\\6,128};

% Caution (left)
\fill[cauorange!20, draw=cauorange, thick, rounded corners=2pt]
  (\lx, \cauYl) rectangle +(\boxW, \cauHl);
\node[font=\scriptsize\bfseries, text=cauorange, align=center]
  at (\lx + 0.5*\boxW, \cauYl + 0.5*\cauHl)
  {Caution\\4,657};

% Reject (left)
% [LAYOUT] Box height is clamped to 0.50cm. Text is centered at
%          the midpoint of the clamped height, so it sits correctly
%          within the enlarged box boundary.
\fill[rejred!15, draw=rejred, thick, rounded corners=2pt]
  (\lx, \rejYl) rectangle +(\boxW, \rejHl);
\node[font=\scriptsize\bfseries, text=rejred, align=center]
  at (\lx + 0.5*\boxW, \rejYl + 0.5*\rejHl)
  {Reject\\515};

% ============================================================
% DRAW RIGHT BOXES
% ============================================================
% Accept (right)
\fill[accgreen!20, draw=accgreen, thick, rounded corners=2pt]
  (\rx, \accYr) rectangle +(\boxW, \accHr);
\node[font=\scriptsize\bfseries, text=accgreen, align=center]
  at (\rx + 0.5*\boxW, \accYr + 0.5*\accHr)
  {Accept\\6,000};

% Caution (right)
\fill[cauorange!20, draw=cauorange, thick, rounded corners=2pt]
  (\rx, \cauYr) rectangle +(\boxW, \cauHr);
\node[font=\scriptsize\bfseries, text=cauorange, align=center]
  at (\rx + 0.5*\boxW, \cauYr + 0.5*\cauHr)
  {Caution\\1,900};

% Reject (right)
\fill[rejred!15, draw=rejred, thick, rounded corners=2pt]
  (\rx, \rejYr) rectangle +(\boxW, \rejHr);
\node[font=\scriptsize\bfseries, text=rejred, align=center]
  at (\rx + 0.5*\boxW, \rejYr + 0.5*\rejHr)
  {Reject\\2,100};

% ============================================================
% COLUMN HEADERS
% ============================================================
\pgfmathsetmacro{\topY}{\accYl + \accHl + 0.35}
\node[font=\scriptsize\bfseries, text=black, align=center]
  at (\lx + 0.5*\boxW, \topY) {B3-only\\(Config 4)};
\node[font=\scriptsize\bfseries, text=black, align=center]
  at (\rx + 0.5*\boxW, \topY) {Full Stack\\(Config 5)};

% ============================================================
% TOTAL ANNOTATION
% ============================================================
\node[font=\tiny, text=black, align=center]
  at (4.95, \topY + 0.5)
  {1,713 total transitions\\(out of 10,000 messages)};

% ============================================================
% FLOW ARROWS
% ============================================================

% Accept -> Accept (green)
\draw[-{Stealth[length=6pt]}, accgreen, thick]
  (\lx + \boxW, \accYl + 0.6*\accHl)
  .. controls (\lx + \boxW + 1.2, \accYl + 0.6*\accHl)
  and (\rx - 1.2, \accYr + 0.6*\accHr) ..
  (\rx, \accYr + 0.6*\accHr);

% Accept -> Caution (orange)
\draw[-{Stealth[length=6pt]}, cauorange, thick]
  (\lx + \boxW, \accYl + 0.15*\accHl)
  .. controls (\lx + \boxW + 1.5, \accYl + 0.15*\accHl)
  and (\rx - 1.5, \cauYr + 0.8*\cauHr) ..
  (\rx, \cauYr + 0.8*\cauHr);
\path (\lx + \boxW, \accYl + 0.15*\accHl) -- (\rx, \cauYr + 0.8*\cauHr) coordinate[midway] (midAccCau);
\node[fill=white, inner sep=2pt, font=\tiny\bfseries, text=cauorange, align=center]
  at (midAccCau) {128 Accept$\to$Caution\\(7.5\%)};

% Caution -> Caution (orange)
\draw[-{Stealth[length=6pt]}, cauorange, thick]
  (\lx + \boxW, \cauYl + 0.55*\cauHl)
  .. controls (\lx + \boxW + 1.2, \cauYl + 0.55*\cauHl)
  and (\rx - 1.2, \cauYr + 0.55*\cauHr) ..
  (\rx, \cauYr + 0.55*\cauHr);

% Caution -> Reject (red) (Starts at Caution/Yellow, ends at Reject/Red)
\draw[-{Stealth[length=6pt]}, rejred, thick]
  (\lx + \boxW, \cauYl + 0.20*\cauHl)
  .. controls (\lx + \boxW + 1.8, \cauYl + 0.20*\cauHl)
  and (\rx - 1.8, \rejYr + 0.7*\rejHr) ..
  (\rx, \rejYr + 0.7*\rejHr);
\path (\lx + \boxW, \cauYl + 0.20*\cauHl) -- (\rx, \rejYr + 0.7*\rejHr) coordinate[midway] (midCauRej);
\node[fill=white, inner sep=2pt, font=\tiny\bfseries, text=rejred, align=center]
  at (midCauRej) {\textbf{1,585}\\Caution$\to$Reject\\(92.5\%)};

% Reject -> Reject (red)
\draw[-{Stealth[length=6pt]}, rejred, thick]
  (\lx + \boxW, \rejYl + 0.5*\rejHl)
  .. controls (\lx + \boxW + 1.2, \rejYl + 0.5*\rejHl)
  and (\rx - 1.2, \rejYr + 0.3*\rejHr) ..
  (\rx, \rejYr + 0.3*\rejHr);

% ============================================================
% TRANSITION ANNOTATIONS (between columns)
% ============================================================
% Zero direct reversals note
\node[font=\tiny\itshape, text=black, align=center,
      text width=3.5cm]
  at (4.95, -0.45)
  {0 direct Accept$\leftrightarrow$Reject reversals};

\end{tikzpicture}
